\begin{frame}{Inheritance by dilation: the relational sieve}
For a finite prime set $S$, $P=\prod_{p\in S}p$ and $\delta_S=\prod_{p\in S}(1-1/p)$:
\[E_S(x)=C_S(x)-\delta_Sx=-\sum_{d\mid P}\mu(d)\{x/d\}.\]
\[D_pf(x)=f(x/p),\qquad T_p=I-D_p.\]
\begin{block}{Exact update for $p\notin S$}
\[E_{S\cup\{p\}}=T_pE_S,\qquad T_pT_q=T_qT_p.\]
\end{block}
Here $\{y\}=y-\lfloor y\rfloor$. For nonempty $S$, the discrepancy is periodic with period $P$ and mean zero.
\tagline{§3.1: inclusion--exclusion and commuting dilations.}
\end{frame}
\begin{frame}{Mellin transform and the arithmetic interface}
\[P_S(s)=\prod_{p\in S}(1-p^{-s}),\quad F_S(s)=\int_0^\infty E_S(x)x^{-s-1}\,dx.\]
\begin{block}{For $0<\operatorname{Re}s<1$}
\[F_S(s)=\frac{\zeta(s)P_S(s)}s.\]
\end{block}
Finite Euler factors preserve zeros and orders in the critical strip. Using meromorphic continuation, for $\operatorname{Re}s>1$,
\[-\frac{F'_S}{F_S}(s)-\frac1s=\sum_{n\ge1}\Lambda_S(n)n^{-s}.\]
Each incorporation removes exactly the powers of the incorporated prime.
\tagline{§§3.2--3.3.}
\end{frame}
\begin{frame}{Classical smoothed observable and explicit formula}
\[T(X)=\sum_{n\le X}\Lambda(n)(X/n-1)=X\int_1^X\psi(t)t^{-2}\,dt.\]
\[W(X)=T(X)-X\log X+(1+\gamma)X.\]
\begin{block}{Explicit formula}
\[W(X)=\log2\pi-\sum_\rho\frac{X^\rho}{\rho(\rho-1)}-r(X),\quad r(X)=\sum_{k\ge1}\frac{X^{-2k}}{2k(2k+1)}.\]
\end{block}
The sum is absolutely convergent for fixed $X\ge1$. Discrete recovery gives $\Delta^2 T(m)=\Lambda(m)/m$ for integers $m\ge2$.
\tagline{§§3.4--3.7. Verify the contour argument, weights, domain and centering.}
\end{frame}

\begin{frame}{Analytic and functional criteria}
\[\boxed{W(X)=O(\sqrt X)\iff\RH\iff Q_N\to0}\]
Under RH,
\[|W(X)-\log2\pi+r(X)|\le C_0\sqrt X,\quad C_0=2+\gamma-\log4\pi.\]
Equality at $X=1$. The criterion itself is unconditional. In $H=L^2((0,\infty),dt/t^2)$, $e_n(t)=\{t/n\}$ and $\chi=\mathbf1_{[1,\infty)}$,
\[Q(K)=\inf_{(a_n)}\frac1{2\pi}\int_{\mathbb R}\frac{|1+\zeta(s)\sum_{n\in K}a_nn^{-s}|^2}{1/4+u^2}\,du,\quad s=1/2+iu.\]
\tagline{§§3.6, 3.10, 4.3. Check the classical dependencies and squarefree restriction.}
\end{frame}
\begin{frame}{Squarefree inheritance and exact gain}
\[V_{S\cup\{p\}}^{\rm lc}=V_S^{\rm lc}+D_pV_S^{\rm lc},\qquad \dim V_S^{\rm lc}=2^{|S|}.\]
For an increasing exhaustion of the primes, the finite Möbius-supported approximants of Báez-Duarte give
\[\RH\iff Q^{\rm lc}(S_j)\to0.\]

\[Q(V)-Q(V')=\|\Pi_Zr\|^2,\qquad I=\tfrac12\log(1/Q).\]
\end{block}
\tagline{§§4.4, 4.7. The information formula uses the isonormal Gaussian model.}
\end{frame}
